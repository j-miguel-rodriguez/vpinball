# plugin CMake file to be included in main project CMakelists
#
# --- YARG + VPX integration plugin ---
#
# This file, VPinballLib.cpp's 3-line registration patch (see its own
# comment), and this whole directory's existence are the project's
# documented "last resort" exception to its own "don't fork VPX" rule
# (see ~/.../vpx-yarg/CLAUDE.md and docs/ios-platform-plan.md): iOS has
# no dynamic plugin loading at all (Apple's code-signing model), so a
# static registration entry here is the only way to get the plugin into
# the app. Everything else -- the plugin's actual C++ source, the
# NativeAOT-compiled YARG.Core engine, build scripts -- lives entirely
# outside this checkout, in the separate `vpx-yarg` project.
#
# Only BUILD_STATIC is implemented here (unlike the other
# CMakeLists_plugin_*.txt files' dual shared/static branches): this
# fork exists for iOS only, which is always BUILD_STATIC (see
# CMakeLists_plugins.txt). The Linux .so build already has its own,
# fully separate CMake project (yarg-vpx-plugin/CMakeLists.txt in the
# vpx-yarg repo) and deployment path -- reproducing a BUILD_SHARED
# branch here would be unused dead code, not a real second platform.
#
# Assumes the sibling-checkout layout this project actually uses, both
# locally and in CI (see vpx-yarg's docs/04-ios-step1.md): this fork
# lives at <vpx-yarg-root>/external/vpinball-ios, so two levels up from
# CMAKE_SOURCE_DIR is <vpx-yarg-root>.
set(VPX_YARG_ROOT "${CMAKE_SOURCE_DIR}/../..")
set(YARG_PLUGIN_SOURCE_DIR "${VPX_YARG_ROOT}/yarg-vpx-plugin")

set(YARG_PLUGIN_SOURCES
   "${YARG_PLUGIN_SOURCE_DIR}/YargVpxPlugin.cpp"
   "${YARG_PLUGIN_SOURCE_DIR}/NativeEngine.cpp"
   "${YARG_PLUGIN_SOURCE_DIR}/AudioClock.cpp"
   "${YARG_PLUGIN_SOURCE_DIR}/Highway.cpp"
   "${YARG_PLUGIN_SOURCE_DIR}/PartyKeysOverlay.cpp"
   "${YARG_PLUGIN_SOURCE_DIR}/AUv3Host.cpp"
   "${YARG_PLUGIN_SOURCE_DIR}/AUv3Discovery.mm"
   "${YARG_PLUGIN_SOURCE_DIR}/iOSChartPath.mm"
   "${YARG_PLUGIN_SOURCE_DIR}/iOSGuitarDiagnostic.mm"
   "${YARG_PLUGIN_SOURCE_DIR}/iOSMidiInput.mm"
)

# JUCE, module-only: no juce_add_gui_app()/juce_add_plugin(), no
# JUCEApplication, no app wrapper of any kind -- VPX keeps its own app
# lifecycle, build, and main loop entirely. This exists purely to reuse
# juce::AudioPluginFormatManager/AudioUnitPluginFormat/AudioPluginInstance
# (juce_audio_processors) for hosting Minimoog Model D's AUv3 as the
# click track's voice (AUv3Host.h/.cpp, ported from moog-juce's
# HelloWorldJam, where this exact hosting approach -- including the
# AVAudioUnitComponentManager-based discovery workaround in
# AUv3Discovery.mm, since JUCE's own AU discovery misses third-party
# AUv3 app extensions -- is already proven working on real hardware).
# Same pinned tag as that project's own CMakeLists.txt, for parity.
#
# juceaide (JUCE's own build-time code-gen tool, which its CMake machinery
# always builds as a macOS HOST executable regardless of -DPLATFORM=ios --
# a separate, native sub-build, not part of the iOS arm64 cross-compile)
# failed its own first real CI run here: juce_gui_basics.mm's
# createNSWindowSnapshot() calls the legacy CGWindowListCreateImage API
# unconditionally (JUCE 7.0.12 -- the pinned tag actually being built --
# has no ScreenCaptureKit-based alternative at all; that was added in a
# later JUCE release, confirmed by fetching and reading this exact
# pinned tag's real source directly, not a newer local checkout that
# first looked similar but turned out to have a materially different
# function here), and the macOS 26 SDK this CI runner's default Xcode
# bundles marks that API fully obsoleted -- a hard compile error,
# unconditionally, regardless of deployment target. Several earlier
# attempts at fixing this (setting CMAKE_OSX_DEPLOYMENT_TARGET and
# MACOSX_DEPLOYMENT_TARGET; patching a #if condition that, it turned
# out, doesn't exist in this JUCE version at all) went nowhere, in one
# case because CMake's PATCH_COMMAND string-argument handling silently
# mishandled a complex inline bash -c command. patch-juce-screenshot.sh
# (a real file, tested locally against the actual fetched 7.0.12
# source before being trusted here) sidesteps all of that: it stubs the
# obsoleted call out entirely, verifying both before and after that it
# did something. Correctness of the stub is irrelevant -- juceaide
# never actually calls createNSWindowSnapshot() at runtime, only
# compiling it matters.
include(FetchContent)
FetchContent_Declare(
   JUCE
   GIT_REPOSITORY https://github.com/juce-framework/JUCE.git
   GIT_TAG 7.0.12
   GIT_SHALLOW TRUE
   # PATCH_COMMAND's own working directory is the freshly-fetched source
   # root (<name>_SOURCE_DIR), per FetchContent/ExternalProject's own
   # documented behavior.
   PATCH_COMMAND bash "${CMAKE_CURRENT_LIST_DIR}/patch-juce-screenshot.sh"
)
FetchContent_MakeAvailable(JUCE)

if(BUILD_STATIC)
   add_compile_options(-fPIC)

   # AudioClock.cpp includes miniaudio.h, which on iOS internally pulls in
   # AVFoundation.h (its AVAudioEngine-based backend) -- confirmed directly
   # from a real failed build: Foundation's own Objective-C headers
   # (NSObjCRuntime.h etc.) don't parse under plain C++ ("unknown type
   # name 'NSString'", "expected unqualified-id"). Force this one source
   # file to compile as Objective-C++ instead, same as how the app's own
   # apps/macos/main.mm does for its own Apple-framework usage. OBJCXX
   # turned out NOT to already be enabled for this "ios" platform
   # configuration specifically (confirmed by a real "language OBJCXX was
   # requested but was not enabled" CMake configure error) -- main.mm must
   # go through some other, separate CMake path. enable_language() here is
   # idempotent if it's later enabled elsewhere too.
   enable_language(OBJCXX)

   set_source_files_properties("${YARG_PLUGIN_SOURCE_DIR}/AudioClock.cpp" PROPERTIES
      LANGUAGE OBJCXX
   )

   # iOSChartPath.mm (docs/11-ios-native-engine-crash.md): plain Apple
   # Foundation/NSBundle usage to resolve the test chart's real on-device
   # path. .mm is CMake's own default-detected Objective-C++ extension,
   # so this explicit property is strictly redundant -- set anyway to
   # match AudioClock.cpp's own explicit precedent above rather than
   # rely on the implicit default.
   set_source_files_properties("${YARG_PLUGIN_SOURCE_DIR}/iOSChartPath.mm" PROPERTIES
      LANGUAGE OBJCXX
   )

   # iOSGuitarDiagnostic.mm (docs/guitar-input-recon.md, Phase 1):
   # TEMPORARY GameController-layer diagnostic logging. Same .mm
   # default-detection note as iOSChartPath.mm above.
   set_source_files_properties("${YARG_PLUGIN_SOURCE_DIR}/iOSGuitarDiagnostic.mm" PROPERTIES
      LANGUAGE OBJCXX
   )

   # iOSMidiInput.mm (docs/midi-input-recon.md): CoreMIDI-based note
   # input, permanent (not a temporary diagnostic like the two above).
   # Same .mm default-detection note as iOSChartPath.mm above.
   set_source_files_properties("${YARG_PLUGIN_SOURCE_DIR}/iOSMidiInput.mm" PROPERTIES
      LANGUAGE OBJCXX
   )

   # AUv3Discovery.mm: AVAudioUnitComponentManager-based AU discovery
   # (JUCE's own AU discovery misses third-party AUv3 app extensions --
   # see AUv3Host.cpp's own comment). Same .mm default-detection note as
   # iOSChartPath.mm above.
   set_source_files_properties("${YARG_PLUGIN_SOURCE_DIR}/AUv3Discovery.mm" PROPERTIES
      LANGUAGE OBJCXX
   )

   add_library(YargPlugin STATIC ${YARG_PLUGIN_SOURCES})

   set_target_properties(YargPlugin PROPERTIES
      PREFIX ""
      OUTPUT_NAME "plugin-yarg"
   )

   target_include_directories(YargPlugin PUBLIC
      ${CMAKE_SOURCE_DIR}/plugins
      ${CMAKE_SOURCE_DIR}/third-party/include/miniaudio
   )

   # NativeEngine.cpp's iOS branch (docs/06-ios-step3.md in vpx-yarg):
   # calls yarg_* directly (build-time linked against the dylib below,
   # resolved by dyld at launch) instead of dlopen/dlsym -- the macro
   # name (YARG_STATIC_LINK_NATIVE_ENGINE) predates the switch away from
   # a statically-archived .a (docs/11-ios-native-engine-crash.md) and
   # really just means "direct-linked" now, not "static", but a global
   # rename wasn't worth the multi-file churn.
   # YargVpxPlugin.cpp's PSC_CLASS_START/PSC_PROP_R/PSC_CLASS_END,
   # UnsubscribeMsg, and SubmitTypeLibrary usage all need this fork's
   # own, newer vpinball API shapes (see that file's own
   # YARG_NEW_VPX_PLUGIN_API comment and docs/07-ios-step4.md) -- this
   # fork is genuinely ahead of the Linux checkout the plugin was first
   # written against, confirmed directly against both real headers.
   target_compile_definitions(YargPlugin PRIVATE
      YARG_STATIC_LINK_NATIVE_ENGINE
      YARG_NEW_VPX_PLUGIN_API
      # Normally set automatically by juce_add_gui_app(PLUGINHOST_AU TRUE)
      # (moog-juce's own CMakeLists.txt) -- set directly here instead,
      # since this project deliberately never calls that app-wrapper
      # function. Only gates JUCE's OWN AU-hosting code into existing;
      # it has no effect on the entitlement AU discovery separately
      # needs (VPinball.entitlements' own inter-app-audio key).
      JUCE_PLUGINHOST_AU=1
      JUCE_WEB_BROWSER=0
      JUCE_USE_CURL=0
      JUCE_DISPLAY_SPLASH_SCREEN=0
   )

   # JUCE module linkage: juce_audio_processors (AudioPluginFormatManager/
   # AudioUnitPluginFormat/AudioPluginInstance, what AUv3Host.cpp wraps)
   # declares a hard module dependency on juce_gui_extra ->
   # juce_gui_basics -> juce_graphics (confirmed directly from each
   # module's own juce_*.h "dependencies:" declaration, not guessed --
   # AudioProcessorEditor is itself a juce::Component, and the AU-hosting
   # code internally needs the native view-wrapping juce_gui_extra
   # provides, even though this project never displays any JUCE-owned
   # window). Linking the top-level module pulls in that whole chain
   # automatically; juce_generate_juce_header() below is what lets
   # AUv3Host.cpp #include <JuceHeader.h> instead of each module header
   # individually.
   target_link_libraries(YargPlugin PRIVATE
      juce::juce_core
      juce::juce_audio_basics
      juce::juce_audio_processors
   )
   juce_generate_juce_header(YargPlugin)

   # yargcore-native-ios.dylib: step 2's own artifact (vpx-yarg's
   # docs/05-ios-step2.md), published via `dotnet publish -r ios-arm64`
   # in yargcore-native-ios/, then had its install name rewritten to
   # @rpath/yargcore-native-ios.dylib by vpx-yarg's own CI step (must
   # happen before this configure step, so this dylib's own recorded -id
   # -- what libvpinball.dylib's build-time link below bakes in as its
   # dependency reference -- is already the correct runtime-relative
   # path, not a build-machine absolute path). Self-contained (SDK's
   # SelfContained=true): unlike the old NativeLib=Static output, this
   # single file already has the NativeAOT runtime pack's PAL support
   # libraries (System.Native, crypto/compression/GSSAPI, GC, etc.)
   # linked in, so no separate PAL .a glob is needed here anymore (see
   # docs/11-ios-native-engine-crash.md for why NativeLib=Static was
   # dropped in the first place -- its own real limitation, not a build
   # mechanics issue: Microsoft's own docs say static libraries aren't
   # officially supported for Native AOT, and that's what crashed
   # yarg_create() silently on real hardware).
   file(GLOB YARGCORE_NATIVE_IOS_DYLIB
      "${VPX_YARG_ROOT}/yargcore-native-ios/bin/Release/net10.0/ios-arm64/publish/yargcore-native-ios.dylib")
   if(NOT YARGCORE_NATIVE_IOS_DYLIB)
      message(WARNING "YargPlugin: yargcore-native-ios.dylib not found under "
         "${VPX_YARG_ROOT}/yargcore-native-ios/bin/ -- run "
         "`dotnet publish -r ios-arm64` there first, or the final app "
         "link will fail with undefined yarg_* symbols.")
   endif()

   # Linking directly against the dylib's path (not a -framework flag) is
   # enough for the linker to resolve yarg_* symbols; actually getting
   # the dylib embedded into the shipped app bundle (so it's there for
   # dyld to load at runtime) is handled separately, by the Xcode
   # project's own existing "Embed Frameworks" run-script build phase
   # (apps/ios/VPinball.xcodeproj), which now also copies this dylib
   # alongside VPinballFramework.framework -- see that script's own
   # comment.
   target_link_libraries(YargPlugin PUBLIC
      ${YARGCORE_NATIVE_IOS_DYLIB}
   )
endif()
